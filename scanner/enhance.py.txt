import cv2
import numpy as np


def enhance_for_ocr(image, mode="auto"):
    if isinstance(image, str):
        image = cv2.imread(image)

    if image is None:
        raise ValueError("Image load nahi hui")

    denoised = cv2.fastNlMeansDenoisingColored(image, None, 5, 5, 7, 21)

    kernel_sharp = np.array([[0, -1, 0],
                              [-1, 5, -1],
                              [0, -1, 0]])
    sharpened = cv2.filter2D(denoised, -1, kernel_sharp)

    if mode == "color":
        return sharpened

    gray = cv2.cvtColor(sharpened, cv2.COLOR_BGR2GRAY)

    if mode == "grayscale":
        return gray

    clahe = cv2.createCLAHE(clipLimit=2.5, tileGridSize=(8, 8))
    enhanced = clahe.apply(gray)

    if mode == "auto":
        return enhanced

    if mode == "bw":
        bw = cv2.adaptiveThreshold(
            enhanced, 255,
            cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
            cv2.THRESH_BINARY, 31, 10
        )
        bw = cv2.medianBlur(bw, 3)
        return bw

    return enhanced


def auto_rotate(image):
    try:
        import pytesseract
        from PIL import Image

        if isinstance(image, np.ndarray):
            pil = Image.fromarray(cv2.cvtColor(image, cv2.COLOR_BGR2RGB))
        else:
            pil = Image.open(image)
            image = cv2.imread(image)

        osd = pytesseract.image_to_osd(pil)
        angle = int([l for l in osd.split("\n") if "Rotate" in l][0].split(":")[1])

        if angle != 0:
            (h, w) = image.shape[:2]
            center = (w // 2, h // 2)
            M = cv2.getRotationMatrix2D(center, angle, 1.0)
            rotated = cv2.warpAffine(image, M, (w, h),
                                      flags=cv2.INTER_CUBIC,
                                      borderMode=cv2.BORDER_REPLICATE)
            return rotated
    except Exception:
        pass
    return image
